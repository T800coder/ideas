\section{Problem}
Lithium diffusion inside a spherical particle is studied using a PDE model. The concentration inside the particle varies with radius and time, and lithium enters or leaves only through the surface. The outputs looked at are the surface concentration $C_s$, the average concentration $C_\avg$, and the concentration profile $C(r)$ inside the particle. Different flux inputs and different diffusivity values are applied to the same particle to see how they change the response. Six experiments are reported. Experiments 1 to 4 change one thing at a time (24 PDE simulations). Experiments 5 and 6 build two-equation models for $C_s$ and $C_\avg$, by the Galerkin method and by SINDy, and test them against further PDE simulations.

%=====================================================================
\section{Model}
The model is Fick's second law in spherical coordinates, written in dimensionless form. The scalings used are
\begin{equation}
  \tau = \frac{D_0 t}{R^2},\qquad r=\frac{\tilde r}{R},\qquad C=\frac{c}{c_0},\qquad f(C)=\frac{D(c)}{D_0},\qquad \delta = \frac{jR}{D_0c_0},
\end{equation}
so $\tau$ is scaled time, $r$ runs from 0 at the centre to 1 at the surface, $C$ is concentration as a fraction of full, $f(C)$ is the diffusivity relative to the reference value $D_0$, and $\delta$ is the scaled surface flux. The governing equation is
\begin{equation}\label{eq:pde}
  \frac{\partial C}{\partial\tau} = \frac{1}{r^2}\frac{\partial}{\partial r}\left[r^2 f(C)\frac{\partial C}{\partial r}\right],
\end{equation}
with boundary and initial conditions
\begin{align}
  f(C)\left.\frac{\partial C}{\partial r}\right|_{r=1} &= -\delta(\tau),\\
  \left.\frac{\partial C}{\partial r}\right|_{r=0} &= 0,\\
  C(r,0) &= 1.
\end{align}
The two reported quantities are
\begin{equation}
  C_s(\tau) = C(1,\tau),\qquad C_\avg(\tau) = 3\int_0^1 C(r,\tau)\,r^2\,dr.
\end{equation}
Integrating Equation~\eqref{eq:pde} over the particle and using the boundary conditions gives
\begin{equation}\label{eq:massbal}
  \frac{dC_\avg}{d\tau} = -3\,\delta(\tau),
\end{equation}
which ties the average concentration directly to the applied flux.

\subsection{Parameters}
The particle radius is $R=2\times10^{-6}$ m, the maximum concentration is $c_0=51554$ mol m$^{-3}$, and the reference diffusivity is $D_0=2\times10^{-14}$ m$^2$ s$^{-1}$. These give $R^2/D_0=200$ s, so one unit of $\tau$ is 200 seconds. The particle starts full and uniform, $C_0=1$, and each run of Experiments 1 to 4 covers $0\le\tau\le10$, that is 2000 seconds. The reference surface flux is $j_0=4.3\times10^{-6}$ mol m$^{-2}$ s$^{-1}$, which gives $\delta_0=j_0R/(D_0c_0)=0.008341$. A C-rate multiplier of 3 is used, so the flux amplitude in every deterministic profile is
\begin{equation}
  A = 3\,\delta_0 = 0.025022.
\end{equation}
The equations are solved in MATLAB using \texttt{pdepe} in spherical mode on a grid of 1000 radial points and 2000 time points, with relative and absolute tolerances of $10^{-9}$ and a maximum step of $10^{-2}$ in $\tau$. These settings are the same for all 24 runs of Experiments 1 to 4. Experiments 5 and 6 use their own run lengths and time steps, given there.

\subsection{Diffusivity}
Unless stated otherwise the diffusivity is the fitted NMC811 curve used in the solver:
\begin{equation}\label{eq:nmc}
  f(C) = p_4 + (p_1-p_4)\exp\left[-\exp\left(-p_2(C-p_3)\right)\right],
\end{equation}
with $p_1=0.1008$, $p_2=-8.2105$, $p_3=0.6986$ and $p_4=0.0009$. The NMC811 diffusivity is concentration dependent and is used as the default material model. It is low near $C=1$, about 0.0009, and rises to about 0.1 at low $C$. Experiment 3 replaces it with a constant $f$.

\subsection{Flux profiles}
Six deterministic flux shapes are used:
\begin{align}
  \texttt{const}:&\quad \delta(\tau)=A,\\
  \texttt{stepUp}:&\quad \delta(\tau)=A\left[1+\tfrac12\tanh 20(\tau-5)\right],\\
  \texttt{linear}:&\quad \delta(\tau)=A\,\frac{\tau}{10},\\
  \texttt{linearDown}:&\quad \delta(\tau)=A\left(1-\frac{\tau}{10}\right),\\
  \texttt{sin}:&\quad \delta(\tau)=A\sin(\omega\tau),\\
  \texttt{cos}:&\quad \delta(\tau)=A\cos(\omega\tau).
\end{align}
The frequency is $\omega=20$ everywhere except in Experiment 2, where it is swept. Since \texttt{sin} and \texttt{cos} are negative for half of each cycle, they push lithium back into the particle, so $C$ can rise above 1 in those runs. The model puts no upper bound on $C$.

\section{Experiment 5: two-equation model by the Galerkin method}
\begin{itemize}
  \item The aim is a model of two ordinary differential equations, one for $C_\avg$ and one for $C_s$, derived from the PDE by the Galerkin method, and then tested against the PDE.
  \item The diffusivity is constant, $f=1$, throughout this experiment and Experiment 6.
\end{itemize}

\subsection{Starting point: the Galerkin formulas}
\begin{itemize}
  \item The concentration is approximated by $N$ fixed radial shapes $P_k(r)$, each multiplied by a coefficient that depends only on time:
  \begin{equation}
    C(r,\tau)\approx\sum_{k=1}^{N}\alpha_k(\tau)\,P_k(r).
  \end{equation}
  \item The shapes are orthogonal with the spherical weight $r^2$, and the first shape is $P_1=1$:
  \begin{equation}\label{eq:orth}
    \int_0^1 r^2\,P_i(r)\,P_j(r)\,dr = 0\quad(i\neq j),\qquad P_1(r)=1.
  \end{equation}
  \item The Galerkin projection of the PDE onto these shapes gives the following two results, in dimensionless form. They are the starting point of this experiment.
  \begin{equation}\label{eq:galCs}
    \frac{dC_s}{d\tau} = -\delta(\tau)\sum_{i=1}^{N}\frac{P_i(1)^2}{\int_0^1 r^2P_i(r)^2\,dr}
    - \sum_{i=1}^{N}\frac{P_i(1)\int_0^1 r^2 P_i'(r)\, f\!\left(\sum_{k=1}^{N}\alpha_kP_k(r)\right)\left(\sum_{k=1}^{N}\alpha_kP_k'(r)\right)dr}{\int_0^1 r^2P_i(r)^2\,dr},
  \end{equation}
  \begin{equation}\label{eq:galCavg}
    \frac{dC_\avg}{d\tau} = \frac{d\alpha_1}{d\tau} = \frac{-\delta(\tau)}{\int_0^1 r^2\,dr} = -3\,\delta(\tau).
  \end{equation}
  \item $f(\cdot)$ is the diffusivity evaluated at the approximate concentration. With constant $f$, it is taken out of the integral in Equation~\eqref{eq:galCs}.
  \item Equation~\eqref{eq:galCavg} is the exact mass balance, Equation~\eqref{eq:massbal}.
  \item Equation~\eqref{eq:galCs} gives $dC_s/d\tau$ in terms of the coefficients $\alpha_k$. To obtain a model in $C_s$ and $C_\avg$, the shapes are chosen and the $\alpha_k$ are written in terms of $C_s$ and $C_\avg$.
\end{itemize}

\subsection{Choice of shapes, $N=2$}
\begin{itemize}
  \item $P_1=1$. The second shape is taken as $P_2=r^2+c$, the simplest shape that is even in $r$ (so $\partial C/\partial r=0$ at the centre) and can bend.
  \item The constant $c$ is fixed by the orthogonality condition, Equation~\eqref{eq:orth}, between $P_1$ and $P_2$:
  \begin{equation}
    \int_0^1 r^2\cdot1\cdot(r^2+c)\,dr = \frac15+\frac c3 = 0\quad\Rightarrow\quad c=-\frac35,\qquad P_1(r)=1,\qquad P_2(r)=r^2-\frac35.
  \end{equation}
  \item The approximation is $C=\alpha_1+\alpha_2\left(r^2-\frac35\right)$, a parabola in $r$.
\end{itemize}

\subsection{Linking $\alpha_1$, $\alpha_2$ to $C_\avg$ and $C_s$}
\textbf{Average concentration.} Using the definition $C_\avg=3\int_0^1C(r,\tau)\,r^2\,dr$ and the approximation:
\begin{equation}
  C_\avg = 3\left[\alpha_1\int_0^1r^2\,dr+\alpha_2\int_0^1r^2\left(r^2-\tfrac35\right)dr\right] = 3\left[\frac{\alpha_1}{3}+\alpha_2\cdot0\right]=\alpha_1.
\end{equation}
\begin{itemize}
  \item The second integral is zero by the orthogonality condition, since $\int_0^1r^2P_2\,dr=\int_0^1r^2P_1P_2\,dr=0$. So the first coefficient is exactly the average concentration, $\alpha_1=C_\avg$, which is why Equation~\eqref{eq:galCavg} is written as $dC_\avg/d\tau$.
\end{itemize}
\textbf{Surface concentration.} At $r=1$:
\begin{equation}
  C_s = \alpha_1P_1(1)+\alpha_2P_2(1) = \alpha_1+\frac25\,\alpha_2.
\end{equation}
\begin{itemize}
  \item With $\alpha_1=C_\avg$:
  \begin{equation}\label{eq:alphas}
    \alpha_1 = C_\avg,\qquad \alpha_2=\frac52\left(C_s-C_\avg\right).
  \end{equation}
\end{itemize}

\subsection{Applying the formula for $dC_s/d\tau$}
\begin{table}[H]\centering
\begin{tabular}{lccc}\toprule
 & $P_i(1)$ & $\int_0^1r^2P_i^2\,dr$ & $P_i'(r)$\\\midrule
$P_1=1$ & 1 & 1/3 & 0\\
$P_2=r^2-3/5$ & 2/5 & 4/175 & $2r$\\\bottomrule
\end{tabular}
\caption{Values needed in Equation~\eqref{eq:galCs}.}
\end{table}
\begin{itemize}
  \item $\int_0^1r^2P_2^2\,dr=\int_0^1\left(r^6-\frac65r^4+\frac9{25}r^2\right)dr=\frac17-\frac6{25}+\frac3{25}=\frac4{175}$.
  \item \textbf{First sum (flux term):}
  \[
    \frac{P_1(1)^2}{1/3}+\frac{P_2(1)^2}{4/175} = \frac1{1/3}+\frac{4/25}{4/175}=3+7=10.
  \]
  \item \textbf{Second sum (diffusion term).} Since $P_1'=0$, only $i=2$ contributes, and inside the bracket only $k=2$ contributes: $\sum_k\alpha_kP_k'(r)=2r\,\alpha_2$. With constant $f$:
  \[
    \frac{P_2(1)\,f\int_0^1r^2\,(2r)(2r\,\alpha_2)\,dr}{4/175} = \frac{\frac25\cdot\frac45\,f\,\alpha_2}{4/175} = 14\,f\,\alpha_2.
  \]
  \item So Equation~\eqref{eq:galCs} becomes
  \[
    \frac{dC_s}{d\tau} = -10\,\delta-14\,f\,\alpha_2.
  \]
  \item Replace $\alpha_2$ using Equation~\eqref{eq:alphas}: $14f\alpha_2=14f\cdot\frac52(C_s-C_\avg)=35f(C_s-C_\avg)$.
\end{itemize}

\subsection{The two-equation Galerkin model}
\begin{equation}\label{eq:galerkin}
  \boxed{\ \frac{dC_\avg}{d\tau}=-3\,\delta(\tau),\qquad \frac{dC_s}{d\tau}=-35f\left(C_s-C_\avg\right)-10\,\delta(\tau),\qquad C_s(0)=C_\avg(0)=1.\ }
\end{equation}
\begin{itemize}
  \item With $f=1$, written term by term: $dC_s/d\tau=-35\,C_s+35\,C_\avg-10\,\delta$. These three coefficients are compared with SINDy in Experiment 6.
  \item The initial values follow from $C(r,0)=1$: $\alpha_1=1$, $\alpha_2=0$.
  \item Assumptions: constant $f$; the profile is a combination of 1 and $r^2$ at every instant; the particle starts uniform.
  \item Slow-flux check: when $\delta$ changes slowly, $C_s$ and $C_\avg$ move together, so $dC_s/d\tau\approx-3\delta$, which gives $C_s-C_\avg=-\delta/(5f)$.
  \item With $N=2$ the equation for $C_s$ contains only $C_s$, $C_\avg$ and $\delta$. With $N\ge3$ it would also contain $\alpha_3,\dots$, which cannot be written with $C_s$ and $C_\avg$ alone.
\end{itemize}

\subsection{Test against the PDE}\label{sec:exp5test}
\begin{itemize}
  \item The flux is $\delta(\tau)=A\cos(\omega\tau)$ at $\omega=0.5$, 1, 2, 3, 5, 10, 20, 30, 50, 100 and 200.
  \item Each run lasts $\tau_{\max}=\max(10,\ 5\text{ periods})$ with time step $\min(0.005,\ \text{period}/100)$, on the same 1000-point radial grid.
  \item The Galerkin model, Equation~\eqref{eq:galerkin}, is solved on the same time points, starting from $C_s=C_\avg=1$.
\end{itemize}
Two questions had to be settled before the comparison could be read: how to measure the error, and over which part of each run.

\subsubsection{Why a relative error}
The first idea was to look at the plain (absolute) error in $C_s$. Over the last period of each run it rises from about $4\times10^{-7}$ at $\omega=0.5$ to about $8\times10^{-4}$ at $\omega=200$, roughly 2000 times larger. On its own this number says little, because the size of the signal changes too: the swing of $C_s$ (its maximum minus its minimum, the \emph{band} in which the values lie) shrinks from 0.30 to 0.0037, about 80 times smaller.

So at high frequency the error grows while the band it must be compared with shrinks. The relative error divides the error by that band. For a predicted signal $\hat y$ and a reference $y$, sampled at the $n$ points $t_1,\dots,t_n$ of a time \emph{window} $W$,
\begin{equation}\label{eq:rel}
  \mathrm{Rel}_W(\hat y,y) = \frac{\RMS_W(\hat y-y)}{\max_W y-\min_W y},\qquad
  \RMS_W(e)=\sqrt{\frac1n\sum_{k=1}^{n}e(t_k)^2},
\end{equation}
given in \% ($100\times$Rel). Here $\hat y=C_{s,\rm model}$ and $y=C_{s,\rm PDE}$:
\[
  \mathrm{Rel}\ (\%) = 100\times\frac{\RMS\big(C_{s,\rm model}-C_{s,\rm PDE}\big)}{\max C_{s,\rm PDE}-\min C_{s,\rm PDE}}.
\]
The numerator and the denominator always use the \emph{same window}, stated with every result.

\why it measures the error against how much $C_s$ actually moves, so it punishes both effects at once: the growing error and the shrinking band. Between $\omega=0.5$ and 200 it rises about $1.6\times10^5$ times, which shows how quickly the model loses accuracy. Dividing by $C_s$ itself was rejected: $C_s$ stays close to 1, so every error would look tiny.

\decided{this formula is used for every error in this report, in Experiment 6 as well.}

\subsubsection{Why both the whole run and the last period}
\begin{itemize}
  \item \textbf{Whole run.} Every run starts from a uniform particle, and the flux is switched on at full size at $\tau=0$. In a real battery the same happens whenever the current changes suddenly: the start of charging or discharging, or a step in current. Diffusion cannot keep up, a steep layer forms near the surface, and the model must follow this transient. The whole-run error includes it. At low frequency the largest whole-run error (about $9.5\times10^{-4}$) occurs right at the start, near $\tau=0.005$.
  \item \textbf{Last period.} After the start-up the response settles into a repeating pattern. The last-period error measures the model in this settled state.
\end{itemize}
\why the two windows answer two different questions: how well the model follows a sudden change, and how well it follows a steady operation. Comparing them shows how much of the error comes from the start-up. At $\omega=0.5$ the whole-run error is 40 times the last-period error; above $\omega=20$ the two are almost equal, because there the model is inaccurate in every period.

\section{Experiment 6: SINDy}
\begin{itemize}
  \item SINDy (sparse identification of nonlinear dynamics, Brunton, Proctor and Kutz, 2016) finds an equation directly from data. The time derivative of a quantity is written as a combination of candidate terms (the library), and only the terms that are needed are kept.
  \item The aim is to find the equation for $dC_s/d\tau$ from PDE data and compare its coefficients with the Galerkin model of Experiment 5.
  \item Before using it on the PDE, the SINDy code and the method of choosing its settings are checked on the Lorenz system, whose equations are known.
\end{itemize}

\subsection{Method}
\begin{itemize}
  \item Data are collected at many times and stacked: $\dot X=\Theta(X)\,\Xi$, where each column of $\Theta$ is one candidate term evaluated on the data, and $\Xi$ holds the unknown coefficients.
  \item $\Xi$ is found by sequentially thresholded least squares (STLSQ): fit by least squares, set every coefficient smaller than a threshold $\lambda$ to zero, fit again using only the remaining terms, and repeat (10 times). The threshold is applied to the coefficients directly.
\end{itemize}

\subsubsection{Training, validation and test data}
\begin{itemize}
  \item \textbf{Training data}: the data SINDy is fitted on. $E_{train}$ is measured on it.
  \item \textbf{Validation data}: new runs that were not used in the fit. They are used to score every model in the $\lambda$ sweep and to choose $\lambda$. $E_{dv}$ and $E_{sim}$ are measured on them.
  \item \textbf{Test data}: a third set, used \emph{once}, after $\lambda$ has been chosen. It is not used for any choice, so it gives an honest score of the chosen model.
\end{itemize}
\decided{one unseen test set for each problem, used once, after training and validation are complete.}

\subsubsection{The three model errors}
All three use Equation~\eqref{eq:rel}. When there are several variables (for Lorenz $x$, $y$, $z$), Rel is computed for each variable separately, with its own band, and the results are averaged, so every variable counts equally whatever its size.
\begin{itemize}
  \item $E_{train}$: Rel of $\Theta(X)\Xi$ against the derivative it was fitted to, on the training data. It can only go down as more terms are kept, so it cannot tell a real term from an unnecessary one.
  \item $E_{dv}$ (derivative on validation data): Rel of $\Theta(X)\Xi$ against the derivative of new data, with $\Theta$ evaluated at the \emph{true} states of the new data. It asks: given the true state, does the model predict the right rate of change?
  \item $E_{sim}$ (simulation on validation data): the SINDy equations are \emph{solved} from the starting point of the new data, and the solution is compared with the true states. It asks: if the model is used on its own, does it follow the true trajectory?
\end{itemize}
They are written out in full for Lorenz in Section~\ref{sec:lorenz-err} and for the PDE in Section~\ref{sec:pde-err}.

\subsubsection{The $\lambda$ grid}\label{sec:grid}
$\lambda$ is swept on a log scale with \textbf{four values per decade}, $\lambda_k=10^{k/4}$. Each step multiplies or divides $\lambda$ by $10^{0.25}\approx1.78$:
\[
  \dots,\ 0.178,\ 0.316,\ 0.562,\ 1.0,\ 1.78,\ 3.16,\ 5.62,\ 10,\ \dots
\]
So after $\lambda=1$ the next smaller value is 0.562, not 0.99 or 0.9. This is why the selected Lorenz $\lambda$ is ``0.56''. The model itself changes just below $\lambda=1$, because the smallest true Lorenz coefficients are $\pm1$; a finer grid would put the edge of the plateau closer to 1 but would select the same model.
\decided{keep four values per decade.}

\subsection{Check on the Lorenz system}
\why before trusting SINDy on the PDE, where the right answer is not known, it is tested on a system whose equations are known exactly. If it recovers them, the code and the method of choosing its settings can be trusted.

\subsubsection{System, data and derivatives}
\begin{equation}
  \dot x=\sigma(y-x),\qquad \dot y = x(\rho-z)-y,\qquad \dot z=xy-\beta z,\qquad
  \sigma=10,\ \rho=28,\ \beta=8/3,
\end{equation}
starting from $x(0)=-8$, $y(0)=7$, $z(0)=27$. With these values the true equations are $\dot x=-10x+10y$, $\dot y=28x-y-xz$, $\dot z=-2.6667z+xy$: seven terms in total. SINDy has to find these from data alone.
\begin{itemize}
  \item Training data every $\Delta t=0.001$ up to $t=100$ ($10^5$ points). Library: all polynomials in $x,y,z$ up to fifth order (56 terms, from 1 to $z^5$). 10 threshold-and-refit passes.
\end{itemize}
SINDy needs the derivatives $\dot X$. Three ways of getting them are compared:
\begin{align*}
  \text{A (exact):}\quad & \dot X \text{ from the equations},\\
  \text{C2 (2nd-order central):}\quad & \dot X_i \approx \frac{X_{i+1}-X_{i-1}}{2\Delta t}, \qquad \text{error} \propto \Delta t^2,\\
  \text{C4 (4th-order central):}\quad & \dot X_i \approx \frac{-X_{i+2}+8X_{i+1}-8X_{i-1}+X_{i-2}}{12\,\Delta t}, \qquad \text{error}\propto\Delta t^4 .
\end{align*}
For C2 and C4 the one or two points at each end, where the central formula cannot be used, are left out.

\why A is the ideal reference. C2 is what is used on the PDE, where only data are available. C4 was added on the professor's suggestion, to see whether a more accurate derivative changes the result. An earlier case, ``exact derivative plus noise'', was removed: the added term was a systematic change to the data rather than measurement noise, and it did not belong in this check.

\subsubsection{How the error of a model is measured}\label{sec:lorenz-err}
Below, $\dot x_{\rm ref}$ is the derivative of the data, taken by the \emph{same} method as in training (A, C2 or C4), and $\dot x_{\rm SINDy}=\Theta(x,y,z)\,\xi_x$ is the SINDy equation for $\dot x$, evaluated at the true $x,y,z$ of the data; likewise for $y$ and $z$. The errors are given as fractions.

\textbf{Training error.} On the training run:
\[
  E_{train} = \frac13\left[
  \frac{\RMS(\dot x_{\rm SINDy}-\dot x_{\rm ref})}{\max\dot x_{\rm ref}-\min\dot x_{\rm ref}} +
  \frac{\RMS(\dot y_{\rm SINDy}-\dot y_{\rm ref})}{\max\dot y_{\rm ref}-\min\dot y_{\rm ref}} +
  \frac{\RMS(\dot z_{\rm SINDy}-\dot z_{\rm ref})}{\max\dot z_{\rm ref}-\min\dot z_{\rm ref}}\right].
\]

\textbf{Derivative error on validation data.} The same formula on each validation run $j=1,2,3$ (window: the whole run, $0\le t\le20$), then the mean over the three runs:
\[
  E_{dv} = \frac13\sum_{j=1}^{3}\ \frac13\sum_{v\in\{x,y,z\}}
  \frac{\RMS\big(\dot v_{\rm SINDy}-\dot v_{\rm ref}\big)_j}{\big(\max \dot v_{\rm ref}-\min\dot v_{\rm ref}\big)_j}.
\]
So $E_{dv}$ contains the $\dot x$, $\dot y$ and $\dot z$ errors, each divided by its own band, and averaged.

\textbf{Simulation error on validation data.} The SINDy equations are solved from the starting point of validation run $j$, giving $x_{\rm SINDy}(t), y_{\rm SINDy}(t), z_{\rm SINDy}(t)$, and compared with the true $x(t),y(t),z(t)$ over the window $0\le t\le T$:
\[
  E_{sim}(T) = \frac13\sum_{j=1}^{3}\ \frac13\sum_{v\in\{x,y,z\}}
  \frac{\RMS_{[0,T]}\big(v_{\rm SINDy}-v\big)_j}{\big(\max_{[0,T]} v-\min_{[0,T]} v\big)_j}.
\]
$T$ is kept short (0.5, 1, 2) because the Lorenz system is chaotic: over a long window every model, even an excellent one, drifts away from the true trajectory. A model whose solution grows without bound gets $E_{sim}=\infty$.

\textbf{Maximum coefficient error.} Lorenz is the only case where the true coefficients are known, so one more number can be given: the largest of the $56\times3=168$ differences between the SINDy coefficients and the true ones,
\[
  \text{max coef.\ err.} = \max_{i,k} \big|\Xi_{ik,\rm SINDy}-\Xi_{ik,\rm true}\big| ,
\]
where the true $\Xi$ has the 7 Lorenz coefficients and 161 zeros. $E_{dv}$ and $E_{sim}$ say whether the model \emph{behaves} correctly; this number says whether its \emph{coefficients} are correct.

\decided{the main measure for Lorenz is $E_{dv}$: SINDy predicts derivatives, and for Lorenz the target is $\dot x,\dot y,\dot z$. $E_{sim}$ is reported alongside. (For the battery the target is $C_s$, so there the main measure is $E_{sim}$; Section~\ref{sec:pde-err}.)}

A smaller derivative error is therefore not always a better model.
\end{itemize}

\subsubsection{Finding a rule for choosing $\lambda$}
The sweep gives many models. The professor asked how anyone given such a sweep should pick one, in a way that works for other problems too. Four rules were tried, each on $E_{dv}$ and on $E_{sim}(T{=}1)$. Each cell gives the number of terms of the chosen model and, in brackets, how many of them are wrong; the correct answer is 7 (0).
\begin{table}[H]\centering\small

\caption{Terms (wrong terms) of the model picked by each rule.}
\end{table}
\begin{enumerate}
  \item \textbf{R1: the largest $\lambda$ whose error is within 10\% of the smallest error.} The natural first idea: take the most accurate model. It \emph{failed} whenever the derivative was approximate, choosing models with up to 57 wrong terms. Lesson: the smallest error belongs to an overfitted model, because the extra terms absorb the error of the finite-difference derivative.
  \item \textbf{R2: within a factor 10 of the smallest error.} A looser tolerance. It still \emph{failed} with $E_{dv}$ for C2 and C4. Lesson: any rule anchored to the minimum error inherits its overfitting; how much tolerance is enough depends on the case.
  \item \textbf{R3: the largest $\lambda$ whose error is below a fixed level ($10^{-3}$).} It \emph{worked} in every case. Its weakness: the level $10^{-3}$ was chosen by looking at these results, and a different problem, with different units and accuracy, would need a different level.
  \item \textbf{R4: the model just after the largest drop in error (the ``knee'').} It \emph{worked} in every case and needs no level to be chosen: it only looks for where adding terms stops paying off. It is also unaffected by any constant scaling of the error, since it compares drops in $\log_{10}$ of the error.
\end{enumerate}
\why R4 was chosen. It found the correct model in all six combinations, like R3, but it needs no problem-specific number, so it can be carried over to the PDE and to other systems unchanged.

\textbf{A warning from C2.} With $E_{dv}$, the largest drop (3 $\to$ 7 terms) is 4.3 orders of magnitude, but the second largest (19 $\to$ 21 terms, the finite-difference fit described above) is 3.4 orders. R4 picks the right model, but by a smaller margin than for A and C4 (14.1 against 0.6, and 7.9 against 0.6 orders).

\subsubsection{The rule, step by step}
\begin{enumerate}
  \item \textbf{Sweep.} Run STLSQ for $\lambda$ on a log scale (four values per decade), from a value large enough to remove every term down to a value where unnecessary terms appear.
  \item \textbf{Merge.} $\lambda$ values that give the same model are merged into one model with a $\lambda$ range.
  \item \textbf{Score.} Compute each model's error on validation data that were not used to fit it ($E_{dv}$ for Lorenz, $E_{sim}$ for the PDE).
  \item \textbf{Discard unusable models.} Remove models whose solution grows without bound when they are solved (unstable models). The empty model (all coefficients zero) is kept. \emph{This step was added after a failure on the PDE, Section~\ref{sec:pde-select}.}
  \item \textbf{Order.} Arrange the remaining models from largest to smallest $\lambda$.
  \item \textbf{Find the knee.} Between each pair of neighbours compute the drop in error, $\log_{10}E_{\rm before}-\log_{10}E_{\rm after}$, and find the largest drop.
  \item \textbf{Select.} Take the model just after the largest drop.
  \item \textbf{Choose $\lambda$.} Use the largest $\lambda$ of that model's range: it gives the same model with the fewest small coefficients to resolve, so it is the cheapest and least sensitive choice.
  \item \textbf{Test.} Score the selected model once on the test data. Nothing is changed after this step.
\end{enumerate}

\begin{itemize}
  \item $E_{sim}$: the SINDy equation for $C_s$ is solved together with $dC_\avg/d\tau=-3\delta$, from the PDE's $C_s$ and $C_\avg$ at the start of the run, and compared with the PDE's $C_s$ over the \textbf{whole run}:
  \[
    E_{sim}\,(\%) = 100\times\frac{\RMS\big(C_{s,\rm SINDy}-C_{s,\rm PDE}\big)}{\max C_{s,\rm PDE}-\min C_{s,\rm PDE}}\quad\text{(whole run)}.
  \]
  Only $C_s$ is compared: every model gets $C_\avg$ exactly, from the mass balance.
  \item $E_{dv}$: the SINDy right-hand side, evaluated at the PDE's $C_s$, $C_\avg$ and $\delta$, against $dC_s/d\tau$ of the PDE data (2nd-order differences, as in training), whole run:
  \[
    E_{dv}\,(\%) = 100\times\frac{\RMS\big(\xi_1C_s+\xi_2C_\avg+\xi_3\delta - dC_s/d\tau\big)}{\max dC_s/d\tau-\min dC_s/d\tau}.
  \]
  \item $E_{train}$: the same as $E_{dv}$ on the training run.
\end{itemize}
\decided{for the battery the main measure is $E_{sim}$ on $C_s$, over the whole run. The aim of the model is $C_s$: even if $dC_s/d\tau$ is predicted imperfectly, $C_s$ must come out right.}

\subsubsection{Cross-test: trained at one frequency, tested at another}
Every selected model is tested on the cosine run at each $\omega$, over the last period, with two error measures:
\begin{itemize}
  \item $C_s$: Rel (\%) of $C_s$, Equation~\eqref{eq:rel}, last period.
  \item gap: the same RMS error divided by the band of the gap $C_s-C_\avg$ instead. Every model gets $C_\avg$ exactly (mass balance), so the only thing a model can get wrong is the gap. At low $\omega$ the swing of $C_s$ is almost all $C_\avg$, which hides gap errors in the $C_s$ measure.
\end{itemize}

\section{Summary}
\begin{itemize}
  \item Changing the flux shape changes the surface concentration response. The average concentration is set by the total charge applied, so matched pairs with the same total charge end at the same $C_\avg$ but different $C_s$.
  \item Increasing the frequency of an oscillating flux reduces the visible oscillation in $C_s$ and confines the response to a thinner layer near the surface, with the material model unchanged.
  \item Increasing the constant diffusivity makes the radial profile more uniform. The maximum gap $C_\avg-C_s$ falls from 0.0710 at $f=0.01$ to 0.000051 at $f=100$.
  \item The average concentration follows the applied flux and is essentially independent of the diffusivity, as Equation~\eqref{eq:massbal} requires.
  \item \texttt{trueRandom} gives different reproducible excitation signals for different seeds, with surface responses that are similar in character but different in detail.
  \item The Galerkin method with the two shapes 1 and $r^2-3/5$ turns the PDE into two ODEs: $dC_\avg/d\tau=-3\delta$ and $dC_s/d\tau=-35f(C_s-C_\avg)-10\,\delta$. Measured by the relative error over the last period, it agrees with the PDE to 0.0001\% at $\omega=0.5$ and loses accuracy quickly above $\omega\approx10$, reaching 21\% at $\omega=200$.
  \item One error formula is used throughout: the RMS error divided by the band (maximum minus minimum) of the reference over the same window, per variable, then averaged.
  \item On the Lorenz system, the knee rule R4 applied to $E_{dv}$ finds the true 7-term equations with every derivative (exact, 2nd-order, 4th-order) at $\lambda=0.56$, and its errors on three unseen test runs match the validation errors. Rules anchored to the smallest error overfit, because extra library terms can fit the finite-difference error.
  \item On the PDE, with the library $\{C_s,C_\avg,\delta\}$ and R4 on $E_{sim}$, SINDy selects the Galerkin-form model at every frequency, with equal and opposite $C_s$ and $C_\avg$ coefficients. The coefficients match Galerkin at $\omega\le1$ and grow with frequency ($-133$, $+133$, $-17.6$ at $\omega=100$). On the unseen random input the model trained at $\omega=100$ is about 3 times more accurate than Galerkin. But no single set of coefficients works at all frequencies, so the library must be extended.
\end{itemize}

\section{Choices settled, and the next step}
\begin{enumerate}
  \item \textbf{Error formula:} Rel $=\RMS(\text{error})/(\max-\min$ of the reference$)$, both over the same window; for several variables, per variable, then averaged.
  \item \textbf{Main error measure for Lorenz:} $E_{dv}$ (R4 applied to $E_{dv}$); $E_{sim}(T)$ reported alongside.
  \item \textbf{Main error measure for the PDE:} $E_{sim}$ of $C_s$, window = whole run.
  \item \textbf{PDE validation data:} cos, sin late, sin ph and cos ph, 10 periods each, stacked.
  \item \textbf{Test sets:} Lorenz, three more new trajectories; PDE, the random multisine. Each used once, after $\lambda$ is chosen.
  \item \textbf{$\lambda$ grid:} four values per decade.
  \item \textbf{Cross-test error measure:} both $C_s$ and gap.
\end{enumerate}
\textcolor{red}{\textbf{Next step (open):} library expansion, adding weighted averages $C_w = 3\int_0^1 C(r,\tau)\,r^2 w(r)\,dr$ to the library. Which weight functions $w(r)$ to use is to be understood first (and asked to the TA), then implemented.}
